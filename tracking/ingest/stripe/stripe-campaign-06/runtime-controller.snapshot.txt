def ranges(d,j,full):
 for f in j['expected_read_files']:

   assert 1<=r['start']<=r['end']<=f['lines']
   if full:assert r['start']<=end+1,('read gap',f['path'],end,r)
   end=max(end,r['end'])
  if full:assert end==f['lines'],('EOF not read',f['path'],end)
